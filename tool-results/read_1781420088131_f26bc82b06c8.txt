   139→// Helper: check if an adjustment applies to a given month/year
   140→// For multi-month adjustments (durationMonths > 1), the adjustment applies to every month
   141→// from effectiveMonth/effectiveYear for durationMonths months
   142→function isAdjustmentActiveInMonth(adjustment: RentAdjustmentData, month: number, year: number): boolean {
   143→  if (adjustment.status !== 'approved') return false
   144→
   145→  const startDate = new Date(adjustment.effectiveYear, adjustment.effectiveMonth - 1, 1)
   146→  const checkDate = new Date(year, month - 1, 1)
   147→  const endDate = new Date(
   148→    adjustment.effectiveYear + Math.floor((adjustment.effectiveMonth - 1 + adjustment.durationMonths) / 12),
   149→    ((adjustment.effectiveMonth - 1 + adjustment.durationMonths) % 12),
   150→    0 // last day of the end month
   151→  )
   152→
   153→  return checkDate >= startDate && checkDate <= endDate
   154→}

   156→// PHASE 2: Dedup guard for 401 handling — prevent multiple simultaneous signOut calls
   157→let isHandling401 = false
   158→
   159→// API helper — handles 401 by redirecting to login (with dedup guard)
   160→async function apiCall(url: string, options?: RequestInit) {
   161→  const res = await fetch(url, {
   162→    ...options,
   163→    headers: {
   164→      'Content-Type': 'application/json',
   165→      ...options?.headers,
   166→    },
   167→  })
   168→
   169→  // PHASE 2 FIX: On 401 Unauthorized, redirect to login page with dedup guard
   170→  if (res.status === 401) {
   171→    if (!isHandling401 && typeof window !== 'undefined') {
   172→      isHandling401 = true
   173→      try {
   174→        // Use NextAuth signOut to properly clear session
   175→        const { signOut } = await import('next-auth/react')
   176→        await signOut({ callbackUrl: '/' })
   177→      } finally {
   178→        // Reset after a delay to allow future 401s if session expires again
   179→        setTimeout(() => { isHandling401 = false }, 5000)
   180→      }
   181→    }
   182→    throw new Error('Session expired. Please log in again.')
   183→  }
   184→
   185→  // PHASE 2: Handle 409 Conflict (optimistic concurrency failure)
   186→  if (res.status === 409) {
   187→    const data = await res.json().catch(() => ({ error: 'Record was modified by another user' }))
   188→    const error = new Error(data.error || 'Record was modified by another user. Please refresh and try again.')
   189→    ;(error as any).isConflict = true
   190→    ;(error as any).statusCode = 409
   191→    throw error
   192→  }
   193→
   194→  if (!res.ok) {
   195→    const data = await res.json().catch(() => ({ error: 'Request failed' }))
   196→    throw new Error(data.error || `Request failed with status ${res.status}`)
   197→  }
   198→  return res.json()
   199→}

   678→    getDashboardData: () => {
   679→      const { company, tenants, payments, properties, expenses, maintenanceItems, adjustments } = get()
   680→      const now = new Date()
   681→      const currentMonth = now.getMonth() + 1
   682→      const currentYear = now.getFullYear()
   683→
   684→      const activeTenants = tenants.filter(t => isFinanciallyActive(t.status))
   685→      const expectedRevenue = activeTenants.reduce((sum, t) => sum + t.rentAmount, 0)
   686→      const currentMonthPayments = payments.filter(p => p.month === currentMonth && p.year === currentYear)
   687→      const collectedRevenue = currentMonthPayments.reduce((sum, p) => sum + p.amount, 0)
   688→
   689→      // Calculate adjustments for current month (including multi-month adjustments)
   690→      const currentMonthAdjustments = adjustments.filter(a => {
   691→        if (a.status !== 'approved') return false
   692→        return isAdjustmentActiveInMonth(a, currentMonth, currentYear)
   693→      })
   694→      const totalAdjustments = currentMonthAdjustments.reduce((sum, a) => sum + a.amount, 0)
   695→
   696→      const overdueTenants = activeTenants.filter(t => {
   697→        const hasPaid = payments.some(p => p.tenantId === t.id && p.month === currentMonth && p.year === currentYear)
   698→        return !hasPaid
   699→      })
   700→
   701→      const partialTenants = activeTenants.filter(t => {
   702→        const monthPayments = payments.filter(p => p.tenantId === t.id && p.month === currentMonth && p.year === currentYear)
   703→        const totalPaid = monthPayments.reduce((sum, p) => sum + p.amount, 0)
   704→        return totalPaid > 0 && totalPaid < t.rentAmount
   705→      })
   706→
   707→      const overdueAmount = overdueTenants.reduce((sum, t) => sum + t.rentAmount, 0)
   708→      const partialAmount = partialTenants.reduce((sum, t) => {
   709→        const paid = payments.filter(p => p.tenantId === t.id && p.month === currentMonth && p.year === currentYear).reduce((s, p) => s + p.amount, 0)
   710→        return sum + (t.rentAmount - paid)
   711→      }, 0)
   712→
   713→      const totalUnits = properties.reduce((sum, p) => sum + p.totalUnits, 0)
   714→      const occupiedUnits = activeTenants.length
   715→      const occupancyRate = totalUnits > 0 ? Math.round((occupiedUnits / totalUnits) * 100) : 0
   716→
   717→      const monthNames = ['Jan', 'Feb', 'Mar', 'Apr', 'May', 'Jun', 'Jul', 'Aug', 'Sep', 'Oct', 'Nov', 'Dec']
   718→      const chartData: any[] = []
   719→      for (let i = 5; i >= 0; i--) {
   720→        let m = currentMonth - i, y = currentYear
   721→        if (m <= 0) { m += 12; y -= 1 }
   722→        const monthExpected = activeTenants.reduce((sum, t) => sum + t.rentAmount, 0)
   723→        const monthPayments = payments.filter(p => p.month === m && p.year === y)
   724→        const monthCollected = monthPayments.reduce((sum, p) => sum + p.amount, 0)
   725→        chartData.push({ month: monthNames[m - 1], expected: monthExpected, collected: monthCollected })
   726→      }
   727→
   728→      const recentPayments = payments
   729→        .sort((a, b) => new Date(b.date).getTime() - new Date(a.date).getTime())
   730→        .slice(0, 10)
   731→        .map(p => ({
   732→          ...p,
   733→          tenant: tenants.find(t => t.id === p.tenantId) || undefined,
   734→        }))
   735→
   736→      const totalExpenses = expenses.reduce((sum, e) => sum + e.amount, 0)
   737→
   738→      return {
   739→        company,
   740→        stats: {
   741→          expectedRevenue,
   742→          collectedRevenue,
   743→          overdueCount: overdueTenants.length,
   744→          overdueAmount: overdueAmount + partialAmount,
   745→          activeTenants: activeTenants.length,
   746→          totalTenants: tenants.length,
   747→          occupancyRate,
   748→          totalUnits,
   749→          occupiedUnits,
   750→          partialCount: partialTenants.length,
   751→          netProfit: collectedRevenue - totalExpenses,
   752→          totalExpenses,
   753→          totalAdjustments,

   757→          ...t,
   758→          payments: payments.filter(p => p.tenantId === t.id),
   759→          property: properties.find(p => p.id === t.propertyId) || undefined,
   760→        })),
   761→        partialTenants: partialTenants.map(t => ({
   762→          ...t,
   763→          payments: payments.filter(p => p.tenantId === t.id),
   764→          property: properties.find(p => p.id === t.propertyId) || undefined,
   765→        })),

   783→    getReportData: (month, year) => {
   784→      const { tenants, payments, properties, expenses, adjustments } = get()
   785→      const activeTenants = tenants.filter(t => isFinanciallyActive(t.status))
   786→      const expectedRevenue = activeTenants.reduce((sum, t) => sum + t.rentAmount, 0)
   787→      const monthPayments = payments.filter(p => p.month === month && p.year === year)
   788→      const totalRevenue = monthPayments.reduce((sum, p) => sum + p.amount, 0)
   789→
   790→      // Calculate adjustments for this month
   791→      const monthAdjustments = adjustments.filter(a => {
   792→        if (a.status !== 'approved') return false
   793→        return isAdjustmentActiveInMonth(a, month, year)
   794→      })
   795→      const totalAdjustments = monthAdjustments.reduce((sum, a) => sum + a.amount, 0)
   796→
   797→      const monthlyExpenses = expenses.filter(e => {
   798→        const d = new Date(e.date)
   799→        return d.getMonth() + 1 === month && d.getFullYear() === year
   800→      })
   801→      const totalExpenses = monthlyExpenses.reduce((sum, e) => sum + e.amount, 0)
   802→      const totalUnits = properties.reduce((sum, p) => sum + p.totalUnits, 0)
   803→      const occupiedUnits = activeTenants.length
   804→      const occupancyRate = totalUnits > 0 ? Math.round((occupiedUnits / totalUnits) * 100) : 0
   805→      const collectionRate = expectedRevenue > 0 ? Math.round((totalRevenue / expectedRevenue) * 100) : 0
   806→
   807→      const expenseBreakdown: Record<string, number> = {}
   808→      for (const e of monthlyExpenses) {
   809→        expenseBreakdown[e.category] = (expenseBreakdown[e.category] || 0) + e.amount
   810→      }
   811→
   812→      const vacantUnits = totalUnits - occupiedUnits
   813→      const avgRent = activeTenants.length > 0 ? expectedRevenue / activeTenants.length : 0
   814→      const vacancyLoss = vacantUnits * avgRent
   815→
   816→      const unpaidTenants = activeTenants.filter(t => !monthPayments.some(p => p.tenantId === t.id))
   817→      const partialTenants = activeTenants.filter(t => {
   818→        const paid = monthPayments.filter(p => p.tenantId === t.id).reduce((s, p) => s + p.amount, 0)
   819→        return paid > 0 && paid < t.rentAmount
   820→      })
   821→      const unpaidAmount = unpaidTenants.reduce((s, t) => s + t.rentAmount, 0) + partialTenants.reduce((s, t) => {
   822→        const paid = monthPayments.filter(p => p.tenantId === t.id).reduce((sum, p) => sum + p.amount, 0)
   823→        return s + (t.rentAmount - paid)
   824→      }, 0)
   825→
   826→      const trend: any[] = []
   827→      for (let i = 5; i >= 0; i--) {
   828→        let m = month - i, y = year
   829→        if (m <= 0) { m += 12; y -= 1 }
   830→        const tPayments = payments.filter(p => p.month === m && p.year === y)
   831→        const revenue = tPayments.reduce((s, p) => s + p.amount, 0)
   832→        const tExpenses = expenses.filter(e => {
   833→          const d = new Date(e.date)
   834→          return d.getMonth() + 1 === m && d.getFullYear() === y
   835→        }).reduce((s, e) => s + e.amount, 0)
   836→        trend.push({ month: m, year: y, revenue, expenses: tExpenses, profit: revenue - tExpenses })
   837→      }
   838→
   839→      const rentalIncome = totalRevenue
   840→      const otherIncome = 0
   841→      const grossRevenue = rentalIncome + otherIncome
   842→      const badDebt = unpaidAmount
   843→      const grossProfit = grossRevenue - vacancyLoss - badDebt
   844→      const costOfOperations = totalExpenses
   845→      const netIncome = grossProfit - costOfOperations
   846→
   847→      return {
   848→        month, year, totalRevenue, expectedRevenue, totalExpenses,
   849→        profitLoss: totalRevenue - totalExpenses,
   850→        occupancyRate, collectionRate, totalUnits, occupiedUnits,
   851→        expenseBreakdown, monthlyExpenses, trend,
   852→        rentalIncome, otherIncome, grossRevenue, vacancyLoss, badDebt,
   853→        grossProfit, costOfOperations, netIncome,
   854→        totalAdjustments,
   855→        cashCollected: totalRevenue,
   856→        adjustmentTotal: totalAdjustments,
   857→        netRevenue: totalRevenue + totalAdjustments,
   858→      }
   859→    },
