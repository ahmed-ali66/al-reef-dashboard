   122→        tenantName: tenant ? getNameByLang(tenant, lang) : 'Unknown',

   125→        amount: p.amount,

   129→        notes: p.notes || null,

   144→          : new Date(r.reservationDate).toISOString().split('T')[0]

   174→          tenantName: r.prospectName || 'Refund',
   175→          propertyName: property ? getNameByLang(property, lang) : '',
   176→          unitNumber: r.unitNumber || null,
   177→          amount: -r.depositAmount, // negative amount for refund
   178→          time: updatedDate,
