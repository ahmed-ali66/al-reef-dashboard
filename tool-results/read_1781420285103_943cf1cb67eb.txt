   111→    const dayPayments = payments.filter(p => {
   112→      const paymentDate = new Date(p.date).toISOString().split('T')[0]
   113→      return paymentDate === selectedDate

   121→        tenantName: tenant ? getNameByLang(tenant, lang) : 'Unknown',
   122→        propertyName: property?.name || '',
   123→        unitNumber: tenant?.unitNumber || null,
   124→        amount: p.amount,
   125→        time: new Date(p.date).toLocaleTimeString('en-AE', { hour: '2-digit', minute: '2-digit' }),

   127→        isLate: p.isLate || false,
   128→        notes: p.notes || null,
   129→      }
   130→    })

   133→    const dayExpenses = expenses.filter(e => {
   134→      const expenseDate = new Date(e.date).toISOString().split('T')[0]

   136→    })

   141→      category: e.category,

   143→      amount: e.amount,
   144→      vendor: e.vendor,
   145→      time: new Date(e.date).toLocaleTimeString('en-AE', { hour: '2-digit', minute: '2-digit' }),
   146→      building: e.building || null,
   147→      recurring: e.recurring || false,
   148→    }))
   149→

   151→    const totalExp = expenseList.reduce((sum, e) => sum + e.amount, 0)
   152→
   153→    setIncomeItems(income)
   154→    setExpenseItems(expenseList)
   155→    setTotalIncome(totalIn)
   156→    setTotalExpense(totalExp)
   157→  }, [selectedDate, lang])
