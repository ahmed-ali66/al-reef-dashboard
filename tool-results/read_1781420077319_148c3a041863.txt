    82→function getReservationStatusLabel(status: string, lang: Language): string {
    83→  switch (status) {
    84→    case 'pending': return t('pending', lang)
    85→    case 'confirmed': return t('confirmed', lang)
    86→    case 'converted': return t('converted', lang)
    87→    case 'cancelled': return t('cancelled', lang)
    88→    case 'expired': return t('expired', lang)
    89→    default: return status
    90→  }
    91→}
    92→
    93→function getReservationStatusColor(status: string): string {
    94→  switch (status) {
    95→    case 'pending': return 'bg-amber-100 text-amber-800 border-amber-200'
    96→    case 'confirmed': return 'bg-sky-100 text-sky-800 border-sky-200'
    97→    case 'converted': return 'bg-emerald-100 text-emerald-800 border-emerald-200'
    98→    case 'cancelled': return 'bg-red-100 text-red-800 border-red-200'
    99→    case 'expired': return 'bg-gray-100 text-gray-600 border-gray-200'
   100→    default: return 'bg-gray-100 text-gray-600 border-gray-200'
   101→  }
   102→}

   104→function getDepositStatusLabel(status: string, lang: Language): string {

   106→    case 'unpaid': return t('unpaid', lang)

   108→    case 'paid': return t('paid', lang)

   135→function getProspectNameByLang(r: ReservationData, lang: Language): string {
   136→  if (lang === 'ar' && r.prospectNameAr) return r.prospectNameAr
   137→  if (lang === 'bn' && r.prospectNameBn) return r.prospectNameBn
   138→  if (lang === 'ur' && r.prospectNameUr) return r.prospectNameUr
   139→  return r.prospectName
   140→}

   142→// ─── Expiry Warning ────────────────────────────────────────────────────────────
   143→
   144→function getDaysUntilExpiry(expiryDate: string | null): number | null {
   145→  if (!expiryDate) return null
   146→  const now = new Date()
   147→  now.setHours(0, 0, 0, 0)
   148→  const expiry = new Date(expiryDate)
   149→  expiry.setHours(0, 0, 0, 0)
   150→  return Math.ceil((expiry.getTime() - now.getTime()) / (1000 * 60 * 60 * 24))
   151→}
