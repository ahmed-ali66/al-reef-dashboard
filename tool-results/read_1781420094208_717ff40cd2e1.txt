   139→   139→// Helper: check if an adjustment applies to a given month/year
   140→   140→// For multi-month adjustments (durationMonths > 1), the adjustment applies to every month
   141→   141→// from effectiveMonth/effectiveYear for durationMonths months
   142→   142→function isAdjustmentActiveInMonth(adjustment: RentAdjustmentData, month: number, year: number): boolean {
   143→   143→  if (adjustment.status !== 'approved') return false
   144→   144→
   145→   145→  const startDate = new Date(adjustment.effectiveYear, adjustment.effectiveMonth - 1, 1)
   146→   146→  const checkDate = new Date(year, month - 1, 1)
   147→   147→  const endDate = new Date(
   148→   148→    adjustment.effectiveYear + Math.floor((adjustment.effectiveMonth - 1 + adjustment.durationMonths) / 12),
   149→   149→    ((adjustment.effectiveMonth - 1 + adjustment.durationMonths) % 12),
   150→   150→    0 // last day of the end month
   151→   151→  )
   152→   152→
   153→   153→  return checkDate >= startDate && checkDate <= endDate
   154→   154→}

   156→   156→// PHASE 2: Dedup guard for 401 handling — prevent multiple simultaneous signOut calls
   157→   157→let isHandling401 = false
   158→   158→
   159→   159→// API helper — handles 401 by redirecting to login (with dedup guard)
   160→   160→async function apiCall(url: string, options?: RequestInit) {
   161→   161→  const res = await fetch(url, {
   162→   162→    ...options,
   163→   163→    headers: {
   164→   164→      'Content-Type': 'application/json',
   165→   165→      ...options?.headers,
   166→   166→    },
   167→   167→  })
   168→   168→
   169→   169→  // PHASE 2 FIX: On 401 Unauthorized, redirect to login page with dedup guard
   170→   170→  if (res.status === 401) {
   171→   171→    if (!isHandling401 && typeof window !== 'undefined') {
   172→   172→      isHandling401 = true
   173→   173→      try {
   174→   174→        // Use NextAuth signOut to properly clear session
   175→   175→        const { signOut } = await import('next-auth/react')
   176→   176→        await signOut({ callbackUrl: '/' })
   177→   177→      } finally {
   178→   178→        // Reset after a delay to allow future 401s if session expires again
   179→   179→        setTimeout(() => { isHandling401 = false }, 5000)
   180→   180→      }
   181→   181→    }
   182→   182→    throw new Error('Session expired. Please log in again.')
   183→   183→  }
   184→   184→
   185→   185→  // PHASE 2: Handle 409 Conflict (optimistic concurrency failure)
   186→   186→  if (res.status === 409) {
   187→   187→    const data = await res.json().catch(() => ({ error: 'Record was modified by another user' }))
   188→   188→    const error = new Error(data.error || 'Record was modified by another user. Please refresh and try again.')
   189→   189→    ;(error as any).isConflict = true
   190→   190→    ;(error as any).statusCode = 409
   191→   191→    throw error
   192→   192→  }
   193→   193→
   194→   194→  if (!res.ok) {
   195→   195→    const data = await res.json().catch(() => ({ error: 'Request failed' }))
   196→   196→    throw new Error(data.error || `Request failed with status ${res.status}`)
   197→   197→  }
   198→   198→  return res.json()
   199→   199→}
