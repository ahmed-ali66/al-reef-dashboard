    85→function getReservationStatusLabel(status: string, lang: Language): string {
    86→  switch (status) {
    87→    case 'pending': return t('pending', lang)
    88→    case 'confirmed': return t('confirmed', lang)
    89→    case 'converted': return t('converted', lang)
    90→    case 'cancelled': return t('cancelled', lang)
    91→    case 'expired': return t('expired', lang)
    92→    default: return status
    93→  }
    94→}

    98→    case 'pending': return 'bg-amber-100 text-amber-800 border-amber-200'

   100→    case 'converted': return 'bg-emerald-100 text-emerald-800 border-emerald-200'
   101→    case 'cancelled': return 'bg-red-100 text-red-800 border-red-200'

   103→    default: return 'bg-gray-100 text-gray-600 border-gray-200'

   138→function getProspectNameByLang(r: ReservationData, lang: Language): string {
   139→  if (lang === 'ar' && r.prospectNameAr) return r.prospectNameAr
   140→  if (lang === 'bn' && r.prospectNameBn) return r.prospectNameBn
   141→  if (lang === 'ur' && r.prospectNameUr) return r.prospectNameUr
   142→  return r.prospectName
   143→}

   145→// ─── Expiry Warning ────────────────────────────────────────────────────────────
   146→
   147→function getDaysUntilExpiry(expiryDate: string | null): number | null {
   148→  if (!expiryDate) return null
   149→  const now = new Date()
   150→  now.setHours(0, 0, 0, 0)
   151→  const expiry = new Date(expiryDate)
   152→  expiry.setHours(0, 0, 0, 0)
   153→  return Math.ceil((expiry.getTime() - now.getTime()) / (1000 * 60 * 60 * 24))
   154→}
