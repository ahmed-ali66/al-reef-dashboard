   146→   146→
   147→   147→function getDaysUntilExpiry(expiryDate: string | null): number | null {
   148→   148→  if (!expiryDate) return null
   149→   149→  const now = new Date()
   150→   150→  now.setHours(0, 0, 0, 0)
   151→   151→  const expiry = new Date(expiryDate)
   152→   152→  expiry.setHours(0, 0, 0, 0)
   153→   153→  return Math.ceil((expiry.getTime() - now.getTime()) / (1000 * 60 * 60 * 24))
   154→   154→}
