    12→// GET /api/backup — Export ALL company data as JSON (comprehensive backup v2.0)
    13→// Includes: Company, Properties, Tenants, Payments, Expenses, Maintenance,
    14→// RecurringBills, BillPayments, Reservations, RentAdjustments, Receipts,
    15→// Notifications, ScoreAuditLogs, Users (no passwords), AuditLogs, ResetRequests
    16→export async function GET() {
    17→  try {
    18→    const user = await getAuthUser()
    19→    if (!user) return unauthorizedResponse()
    20→
    21→    // Only owner/admin can create backups
    22→    if (user.role !== 'owner' && user.role !== 'admin') {
    23→      return forbiddenResponse('Only owners and admins can create backups')
    24→    }
    25→
    26→    const companyId = user.companyId
    27→
    28→    // ─── Fetch ALL active data in parallel ───
    29→    const [
    30→      company,
    31→      properties,
    32→      tenants,
    33→      expenses,
    34→      maintenance,
    35→      users,
    36→      auditLogs,
    37→      recurringBills,
    38→      billPayments,
    39→      reservations,
    40→      rentAdjustments,
    41→      receipts,
    42→      notifications,
    43→      scoreAuditLogs,
    44→      resetRequests,
    45→    ] = await Promise.all([
    46→      prisma.company.findUnique({ where: { id: companyId } }),
    47→      prisma.property.findMany({ where: { companyId, deletedAt: null } }),
    48→      prisma.tenant.findMany({
    49→        where: { companyId, deletedAt: null },
    50→        include: { payments: true, adjustments: true, scoreAuditLogs: true },
    51→      }),
    52→      prisma.expense.findMany({ where: { companyId, deletedAt: null } }),
    53→      prisma.maintenance.findMany({ where: { companyId, deletedAt: null } }),
    54→      prisma.user.findMany({
    55→        where: { companyId },
    56→        select: {
    57→          id: true, email: true, name: true, nameAr: true, nameBn: true, nameUr: true,
    58→          role: true, isActive: true, mustChangePassword: true, twoFactorEnabled: true,
    59→          createdAt: true, updatedAt: true,
    60→          // Exclude password and 2FA secrets for security
    61→        },
    62→      }),
    63→      prisma.auditLog.findMany({
    64→        where: { companyId },
    65→        orderBy: { createdAt: 'desc' },
    66→        take: 5000,
    67→      }),
    68→      // ─── NEW: Recurring Bills with payments ───
    69→      prisma.recurringBill.findMany({
    70→        where: { companyId, deletedAt: null },
    71→        include: { payments: { orderBy: { paymentDate: 'desc' } } },
    72→      }),
    73→      // ─── NEW: Standalone Bill Payments (for cross-reference) ───
    74→      prisma.billPayment.findMany({
    75→        where: { companyId },
    76→        orderBy: { paymentDate: 'desc' },
    77→      }),
    78→      // ─── NEW: Reservations ───
    79→      prisma.reservation.findMany({
    80→        where: { companyId, deletedAt: null },
    81→      }),
    82→      // ─── NEW: Rent Adjustments ───
    83→      prisma.rentAdjustment.findMany({
    84→        where: { companyId },
    85→      }),
    86→      // ─── NEW: Receipts ───
    87→      prisma.receipt.findMany({
    88→        where: { companyId },
    89→      }),
    90→      // ─── NEW: Notifications ───
    91→      prisma.notification.findMany({
    92→        where: { companyId },
    93→        orderBy: { createdAt: 'desc' },
    94→        take: 5000,
    95→      }),
    96→      // ─── NEW: Score Audit Logs ───
    97→      prisma.scoreAuditLog.findMany({
    98→        where: { companyId },
    99→        orderBy: { createdAt: 'desc' },
   100→      }),
   101→      // ─── NEW: Reset Requests ───
   102→      prisma.resetRequest.findMany({
   103→        where: { companyId },
   104→        orderBy: { createdAt: 'desc' },
   105→        take: 500,
   106→      }),
   107→    ])
   108→
   109→    // ─── Fetch ALL soft-deleted data ───
   110→    const [
   111→      deletedProperties,
   112→      deletedTenants,
   113→      deletedExpenses,
   114→      deletedMaintenance,
   115→      deletedRecurringBills,
   116→      deletedReservations,
   117→    ] = await Promise.all([
   118→      prisma.property.findMany({ where: { companyId, deletedAt: { not: null } } }),
   119→      prisma.tenant.findMany({
   120→        where: { companyId, deletedAt: { not: null } },
   121→        include: { payments: true },
   122→      }),
   123→      prisma.expense.findMany({ where: { companyId, deletedAt: { not: null } } }),
   124→      prisma.maintenance.findMany({ where: { companyId, deletedAt: { not: null } } }),
   125→      prisma.recurringBill.findMany({
   126→        where: { companyId, deletedAt: { not: null } },
   127→        include: { payments: true },
   128→      }),
   129→      prisma.reservation.findMany({
   130→        where: { companyId, deletedAt: { not: null } },
   131→      }),
   132→    ])
   133→
   134→    const backup = {
   135→      version: '2.0',
   136→      exportedAt: new Date().toISOString(),
   137→      exportedBy: user.email,
   138→      company,
   139→      data: {
   140→        properties,
   141→        tenants,
   142→        expenses,
   143→        maintenance,
   144→        users,
   145→        auditLogs,
   146→        // v2.0 additions
   147→        recurringBills,
   148→        billPayments,
   149→        reservations,
   150→        rentAdjustments,
   151→        receipts,
   152→        notifications,
   153→        scoreAuditLogs,
   154→        resetRequests,
   155→      },
   156→      deleted: {
   157→        properties: deletedProperties,
   158→        tenants: deletedTenants,
   159→        expenses: deletedExpenses,
   160→        maintenance: deletedMaintenance,
   161→        recurringBills: deletedRecurringBills,
   162→        reservations: deletedReservations,
   163→      },
   164→    }
   165→
   166→    const backupJson = JSON.stringify(backup, null, 2)
   167→    const backupSize = Buffer.byteLength(backupJson, 'utf-8')
   168→    const recordCount =
   169→      properties.length + tenants.length + expenses.length + maintenance.length +
   170→      users.length + recurringBills.length + billPayments.length +
   171→      reservations.length + rentAdjustments.length + receipts.length
   172→
   173→    // Compute SHA-256 data hash for integrity verification
   174→    const dataHash = crypto.createHash('sha256').update(backupJson).digest('hex')
   175→
   176→    // Create BackupRecord
   177→    await prisma.backupRecord.create({
   178→      data: {
   179→        companyId,
   180→        type: 'manual',
   181→        size: backupSize,
   182→        recordCount,
   183→        status: 'completed',
   184→        dataHash,
   185→        triggeredBy: user.id,
   186→      },
   187→    })
   188→
   189→    // Log the backup
   190→    await createAuditLog({
   191→      action: 'BACKUP',
   192→      entity: 'Company',
   193→      entityId: companyId,
   194→      userId: user.id,
   195→      companyId,
   196→      details: {
   197→        version: '2.0',
   198→        properties: properties.length,
   199→        tenants: tenants.length,
   200→        expenses: expenses.length,
   201→        maintenance: maintenance.length,
   202→        users: users.length,
   203→        recurringBills: recurringBills.length,
   204→        billPayments: billPayments.length,
   205→        reservations: reservations.length,
   206→        rentAdjustments: rentAdjustments.length,
   207→        receipts: receipts.length,
   208→        notifications: notifications.length,
   209→        dataHash,
   210→      },
   211→    })
   212→
   213→    return new Response(backupJson, {
   214→      headers: {
   215→        'Content-Type': 'application/json',
   216→        'Content-Disposition': `attachment; filename="al-reef-backup-${new Date().toISOString().split('T')[0]}.json"`,
   217→        'X-Backup-Hash': dataHash,
   218→        'X-Backup-Version': '2.0',
   219→      },
   220→    })
   221→  } catch (error) {
   222→    console.error('Backup error:', error)
   223→    return errorResponse('Failed to create backup', 500)
   224→  }
   225→}

   227→// POST /api/backup — Restore data from a backup file (v1.0 and v2.0 compatible)
   228→export async function POST(request: Request) {
   229→  try {
   230→    const user = await getAuthUser()
   231→    if (!user) return unauthorizedResponse()
   232→
   233→    // Only admin can restore backups
   234→    if (user.role !== 'admin') {
   235→      return forbiddenResponse('Only admins can restore backups')
   236→    }
   237→
   238→    const body = await request.json()
   239→
   240→    if (!body.version || !body.company || !body.data) {
   241→      return errorResponse('Invalid backup file format')
   242→    }
   243→
   244→    const companyId = user.companyId
   245→
   246→    // Verify the backup belongs to this company
   247→    if (body.company.id !== companyId) {
   248→      return errorResponse('Backup file belongs to a different company')
   249→    }
   250→
   251→    // Log the restore attempt
   252→    await createAuditLog({
   253→      action: 'RESTORE_START',
   254→      entity: 'Company',
   255→      entityId: companyId,
   256→      userId: user.id,
   257→      companyId,
   258→      details: { backupDate: body.exportedAt, version: body.version },
   259→    })
   260→
   261→    const summary: Record<string, number> = {
   262→      properties: 0, tenants: 0, payments: 0, expenses: 0, maintenance: 0,
   263→      recurringBills: 0, billPayments: 0, reservations: 0, rentAdjustments: 0,
   264→      receipts: 0, notifications: 0, scoreAuditLogs: 0, resetRequests: 0,
   265→      deletedProperties: 0, deletedTenants: 0, deletedExpenses: 0,
   266→      deletedMaintenance: 0, deletedRecurringBills: 0, deletedReservations: 0,
   267→    }
   268→
   269→    // ─── Upsert Properties ───
   270→    if (body.data.properties) {
   271→      for (const prop of body.data.properties) {
   272→        await prisma.property.upsert({
   273→          where: { id: prop.id },
   274→          update: {
   275→            name: prop.name, nameAr: prop.nameAr, nameBn: prop.nameBn, nameUr: prop.nameUr,
   276→            type: prop.type, address: prop.address, totalUnits: prop.totalUnits, floors: prop.floors,
   277→            archived: prop.archived,
   278→          },
   279→          create: {
   280→            id: prop.id, companyId,
   281→            name: prop.name, nameAr: prop.nameAr, nameBn: prop.nameBn, nameUr: prop.nameUr,
   282→            type: prop.type, address: prop.address, totalUnits: prop.totalUnits, floors: prop.floors,
   283→            archived: prop.archived,
   284→          },
   285→        })
   286→        summary.properties++
   287→      }
   288→    }
   289→
   290→    // ─── Upsert Tenants (with Phase 1 fields) ───
   291→    if (body.data.tenants) {
   292→      for (const tenant of body.data.tenants) {
   293→        await prisma.tenant.upsert({
   294→          where: { id: tenant.id },
   295→          update: {
   296→            name: tenant.name, phone: tenant.phone, rentAmount: tenant.rentAmount,
   297→            status: tenant.status, unitNumber: tenant.unitNumber,
   298→            openingBalance: tenant.openingBalance ?? 0,
   299→            creditBalance: tenant.creditBalance ?? 0,
   300→            legalCase: tenant.legalCase ?? false,
   301→            legalCaseNumber: tenant.legalCaseNumber ?? null,
   302→            legalCaseNotes: tenant.legalCaseNotes ?? null,
   303→          },
   304→          create: {
   305→            id: tenant.id, companyId, propertyId: tenant.propertyId,
   306→            name: tenant.name, nameAr: tenant.nameAr, nameBn: tenant.nameBn, nameUr: tenant.nameUr,
   307→            phone: tenant.phone, whatsapp: tenant.whatsapp, email: tenant.email,
   308→            emiratesId: tenant.emiratesId, nationality: tenant.nationality,
   309→            employer: tenant.employer, emergencyContact: tenant.emergencyContact,
   310→            unitNumber: tenant.unitNumber, unitType: tenant.unitType,
   311→            floor: tenant.floor, sizeSqft: tenant.sizeSqft,
   312→            rentAmount: tenant.rentAmount, municipalityFee: tenant.municipalityFee,
   313→            securityDeposit: tenant.securityDeposit, paymentMethod: tenant.paymentMethod,
   314→            leaseStart: tenant.leaseStart ? new Date(tenant.leaseStart) : null,
   315→            leaseEnd: tenant.leaseEnd ? new Date(tenant.leaseEnd) : null,
   316→            contractDuration: tenant.contractDuration, renewalStatus: tenant.renewalStatus,
   317→            newRent: tenant.newRent, status: tenant.status || 'active',
   318→            latePaymentCount: tenant.latePaymentCount || 0,
   319→            tenantScore: tenant.tenantScore ?? 100,
   320→            systemScore: tenant.systemScore ?? 100,
   321→            notes: tenant.notes,
   322→            openingBalance: tenant.openingBalance ?? 0,
   323→            creditBalance: tenant.creditBalance ?? 0,
   324→            legalCase: tenant.legalCase ?? false,
   325→            legalCaseNumber: tenant.legalCaseNumber ?? null,
   326→            legalCaseNotes: tenant.legalCaseNotes ?? null,
   327→          },
   328→        })
   329→        summary.tenants++
   330→
   331→        // Upsert payments
   332→        if (tenant.payments) {
   333→          for (const payment of tenant.payments) {
   334→            await prisma.payment.upsert({
   335→              where: { id: payment.id },
   336→              update: { amount: payment.amount, date: new Date(payment.date), month: payment.month, year: payment.year },
   337→              create: {
   338→                id: payment.id, tenantId: tenant.id, companyId,
   339→                amount: payment.amount, date: new Date(payment.date),
   340→                month: payment.month, year: payment.year,
   341→                method: payment.method, reference: payment.reference,
   342→                receiptNumber: payment.receiptNumber, notes: payment.notes,
   343→                isLate: payment.isLate || false, daysLate: payment.daysLate || 0,
   344→                allocationType: payment.allocationType || 'CURRENT_RENT',
   345→              },
   346→            })
   347→            summary.payments++
   348→          }
   349→        }
   350→
   351→        // Upsert score audit logs (v2.0)
   352→        if (tenant.scoreAuditLogs) {
   353→          for (const log of tenant.scoreAuditLogs) {
   354→            await prisma.scoreAuditLog.upsert({
   355→              where: { id: log.id },
   356→              update: { previousScore: log.previousScore, newScore: log.newScore, changeType: log.changeType },
   357→              create: {
   358→                id: log.id, tenantId: tenant.id, companyId,
   359→                previousScore: log.previousScore, newScore: log.newScore,
   360→                changeType: log.changeType, changedBy: log.changedBy,
   361→                changedById: log.changedById, reason: log.reason,
   362→              },
   363→            })
   364→            summary.scoreAuditLogs++
   365→          }
   366→        }
   367→      }
   368→    }
   369→
   370→    // ─── Upsert Expenses ───
   371→    if (body.data.expenses) {
   372→      for (const expense of body.data.expenses) {
   373→        await prisma.expense.upsert({
   374→          where: { id: expense.id },
   375→          update: { category: expense.category, description: expense.description, amount: expense.amount },
   376→          create: {
   377→            id: expense.id, companyId,
   378→            category: expense.category, description: expense.description,
   379→            amount: expense.amount, date: new Date(expense.date),
   380→            vendor: expense.vendor, invoiceNumber: expense.invoiceNumber,
   381→            recurring: expense.recurring || false, building: expense.building,
   382→          },
   383→        })
   384→        summary.expenses++
   385→      }
   386→    }
   387→
   388→    // ─── Upsert Maintenance ───
   389→    if (body.data.maintenance) {
   390→      for (const maint of body.data.maintenance) {
   391→        await prisma.maintenance.upsert({
   392→          where: { id: maint.id },
   393→          update: { title: maint.title, status: maint.status },
   394→          create: {
   395→            id: maint.id, companyId, propertyId: maint.propertyId,
   396→            title: maint.title, description: maint.description,
   397→            category: maint.category, vendor: maint.vendor,
   398→            priority: maint.priority || 'medium', status: maint.status || 'pending',
   399→            estimatedCost: maint.estimatedCost, actualCost: maint.actualCost,
   400→            completedAt: maint.completedAt ? new Date(maint.completedAt) : null,
   401→          },
   402→        })
   403→        summary.maintenance++
   404→      }
   405→    }
   406→
   407→    // ─── v2.0: Upsert Recurring Bills ───
   408→    if (body.data.recurringBills) {
   409→      for (const bill of body.data.recurringBills) {
   410→        await prisma.recurringBill.upsert({
   411→          where: { id: bill.id },
   412→          update: {
   413→            providerName: bill.providerName, serviceType: bill.serviceType,
   414→            currentOutstanding: bill.currentOutstanding,
   415→            previousOutstanding: bill.previousOutstanding,
   416→            totalAmountDue: bill.totalAmountDue,
   417→            nextDueDate: new Date(bill.nextDueDate),
   418→            status: bill.status,
   419→          },
   420→          create: {
   421→            id: bill.id, companyId, propertyId: bill.propertyId,
   422→            providerName: bill.providerName, serviceType: bill.serviceType,
   423→            accountNumber: bill.accountNumber, contractNumber: bill.contractNumber,
   424→            currentOutstanding: bill.currentOutstanding,
   425→            previousOutstanding: bill.previousOutstanding ?? 0,
   426→            totalAmountDue: bill.totalAmountDue,
   427→            lastPaymentAmount: bill.lastPaymentAmount,
   428→            lastPaymentDate: bill.lastPaymentDate ? new Date(bill.lastPaymentDate) : null,
   429→            nextDueDate: new Date(bill.nextDueDate),
   430→            billingFrequency: bill.billingFrequency,
   431→            autoRenew: bill.autoRenew ?? true,
   432→            gracePeriodDays: bill.gracePeriodDays ?? 0,
   433→            status: bill.status || 'active',
   434→            notes: bill.notes,
   435→            attachmentUrls: bill.attachmentUrls,
   436→            buildingName: bill.buildingName,
   437→            ownerName: bill.ownerName,
   438→            propertyManager: bill.propertyManager,
   439→          },
   440→        })
   441→        summary.recurringBills++
   442→
   443→        // Upsert bill payments
   444→        if (bill.payments) {
   445→          for (const bp of bill.payments) {
   446→            await prisma.billPayment.upsert({
   447→              where: { id: bp.id },
   448→              update: { amount: bp.amount, paymentDate: new Date(bp.paymentDate) },
   449→              create: {
   450→                id: bp.id, companyId, recurringBillId: bill.id,
   451→                amount: bp.amount, paymentDate: new Date(bp.paymentDate),
   452→                paymentMethod: bp.paymentMethod, reference: bp.reference,
   453→                notes: bp.notes, outstandingBefore: bp.outstandingBefore,
   454→                outstandingAfter: bp.outstandingAfter, createdBy: bp.createdBy,
   455→              },
   456→            })
   457→            summary.billPayments++
   458→          }
   459→        }
   460→      }
   461→    }
   462→
   463→    // ─── v2.0: Upsert Standalone Bill Payments (for bills that may be soft-deleted) ───
   464→    if (body.data.billPayments && !body.data.recurringBills) {
   465→      for (const bp of body.data.billPayments) {
   466→        await prisma.billPayment.upsert({
   467→          where: { id: bp.id },
   468→          update: { amount: bp.amount, paymentDate: new Date(bp.paymentDate) },
   469→          create: {
   470→            id: bp.id, companyId, recurringBillId: bp.recurringBillId,
   471→            amount: bp.amount, paymentDate: new Date(bp.paymentDate),
   472→            paymentMethod: bp.paymentMethod, reference: bp.reference,
   473→            notes: bp.notes, outstandingBefore: bp.outstandingBefore,
   474→            outstandingAfter: bp.outstandingAfter, createdBy: bp.createdBy,
   475→          },
   476→        })
   477→        summary.billPayments++
   478→      }
   479→    }
   480→
   481→    // ─── v2.0: Upsert Reservations ───
   482→    if (body.data.reservations) {
   483→      for (const res of body.data.reservations) {
   484→        await prisma.reservation.upsert({
   485→          where: { id: res.id },
   486→          update: { status: res.status, depositStatus: res.depositStatus },
   487→          create: {
   488→            id: res.id, companyId, propertyId: res.propertyId,
   489→            unitNumber: res.unitNumber, prospectName: res.prospectName,
   490→            prospectNameAr: res.prospectNameAr, prospectNameBn: res.prospectNameBn,
   491→            prospectNameUr: res.prospectNameUr, prospectPhone: res.prospectPhone,
   492→            prospectWhatsapp: res.prospectWhatsapp, prospectEmail: res.prospectEmail,
   493→            reservationDate: new Date(res.reservationDate),
   494→            expectedMoveInDate: res.expectedMoveInDate ? new Date(res.expectedMoveInDate) : null,
   495→            expiryDate: res.expiryDate ? new Date(res.expiryDate) : null,
   496→            depositAmount: res.depositAmount, depositStatus: res.depositStatus || 'unpaid',
   497→            depositPaymentMethod: res.depositPaymentMethod,
   498→            depositReference: res.depositReference,
   499→            status: res.status || 'pending',
   500→            convertedTenantId: res.convertedTenantId,
   501→            depositAppliedTo: res.depositAppliedTo,
   502→            depositAppliedAmount: res.depositAppliedAmount,
   503→            notes: res.notes,
   504→          },
   505→        })
   506→        summary.reservations++
   507→      }
   508→    }
   509→
   510→    // ─── v2.0: Upsert Rent Adjustments ───
   511→    if (body.data.rentAdjustments) {
   512→      for (const adj of body.data.rentAdjustments) {
   513→        await prisma.rentAdjustment.upsert({
   514→          where: { id: adj.id },
   515→          update: { amount: adj.amount, status: adj.status },
   516→          create: {
   517→            id: adj.id, companyId, tenantId: adj.tenantId, propertyId: adj.propertyId,
   518→            amount: adj.amount, adjustmentType: adj.adjustmentType,
   519→            reason: adj.reason, notes: adj.notes,
   520→            effectiveMonth: adj.effectiveMonth, effectiveYear: adj.effectiveYear,
   521→            durationMonths: adj.durationMonths || 1,
   522→            status: adj.status || 'approved', createdBy: adj.createdBy,
   523→          },
   524→        })
   525→        summary.rentAdjustments++
   526→      }
   527→    }
   528→
   529→    // ─── v2.0: Upsert Receipts ───
   530→    if (body.data.receipts) {
   531→      for (const rec of body.data.receipts) {
   532→        await prisma.receipt.upsert({
   533→          where: { id: rec.id },
   534→          update: { amount: rec.amount },
   535→          create: {
   536→            id: rec.id, companyId, tenantId: rec.tenantId,
   537→            paymentId: rec.paymentId, receiptNumber: rec.receiptNumber,
   538→            amount: rec.amount, date: new Date(rec.date),
   539→            month: rec.month, year: rec.year,
   540→            description: rec.description, createdBy: rec.createdBy,
   541→          },
   542→        })
   543→        summary.receipts++
   544→      }
   545→    }
   546→
   547→    // ─── v2.0: Upsert Notifications ───
   548→    if (body.data.notifications) {
   549→      for (const notif of body.data.notifications) {
   550→        await prisma.notification.upsert({
   551→          where: { id: notif.id },
   552→          update: { read: notif.read },
   553→          create: {
   554→            id: notif.id, companyId,
   555→            userId: notif.userId, type: notif.type,
   556→            title: notif.title, message: notif.message,
   557→            data: notif.data, read: notif.read || false,
   558→          },
   559→        })
   560→        summary.notifications++
   561→      }
   562→    }
   563→
   564→    // ─── v2.0: Upsert Reset Requests ───
   565→    if (body.data.resetRequests) {
   566→      for (const req of body.data.resetRequests) {
   567→        await prisma.resetRequest.upsert({
   568→          where: { id: req.id },
   569→          update: { status: req.status },
   570→          create: {
   571→            id: req.id, email: req.email, name: req.name,
   572→            message: req.message, status: req.status || 'pending',
   573→            companyId: req.companyId,
   574→          },
   575→        })
   576→        summary.resetRequests++
   577→      }
   578→    }
   579→
   580→    // ─── Restore soft-deleted records ───
   581→    if (body.deleted?.properties) {
   582→      for (const prop of body.deleted.properties) {
   583→        await prisma.property.upsert({
   584→          where: { id: prop.id },
   585→          update: { name: prop.name, deletedAt: prop.deletedAt ? new Date(prop.deletedAt) : null },
   586→          create: {
   587→            id: prop.id, companyId, name: prop.name, nameAr: prop.nameAr,
   588→            type: prop.type, address: prop.address, totalUnits: prop.totalUnits,
   589→            floors: prop.floors, archived: prop.archived,
   590→            deletedAt: prop.deletedAt ? new Date(prop.deletedAt) : null,
   591→          },
   592→        })
   593→        summary.deletedProperties++
   594→      }
   595→    }
   596→
   597→    if (body.deleted?.tenants) {
   598→      for (const tenant of body.deleted.tenants) {
   599→        await prisma.tenant.upsert({
   600→          where: { id: tenant.id },
   601→          update: { name: tenant.name, phone: tenant.phone, deletedAt: tenant.deletedAt ? new Date(tenant.deletedAt) : null },
   602→          create: {
   603→            id: tenant.id, companyId, propertyId: tenant.propertyId,
   604→            name: tenant.name, phone: tenant.phone,
   605→            rentAmount: tenant.rentAmount, status: tenant.status || 'inactive',
   606→            deletedAt: tenant.deletedAt ? new Date(tenant.deletedAt) : null,
   607→          },
   608→        })
   609→        summary.deletedTenants++
   610→      }
   611→    }
   612→
   613→    if (body.deleted?.expenses) {
   614→      for (const expense of body.deleted.expenses) {
   615→        await prisma.expense.upsert({
   616→          where: { id: expense.id },
   617→          update: { category: expense.category, amount: expense.amount, deletedAt: expense.deletedAt ? new Date(expense.deletedAt) : null },
   618→          create: {
   619→            id: expense.id, companyId, category: expense.category,
   620→            description: expense.description, amount: expense.amount,
   621→            date: new Date(expense.date), vendor: expense.vendor,
   622→            deletedAt: expense.deletedAt ? new Date(expense.deletedAt) : null,
   623→          },
   624→        })
   625→        summary.deletedExpenses++
   626→      }
   627→    }
   628→
   629→    if (body.deleted?.maintenance) {
   630→      for (const maint of body.deleted.maintenance) {
   631→        await prisma.maintenance.upsert({
   632→          where: { id: maint.id },
   633→          update: { title: maint.title, deletedAt: maint.deletedAt ? new Date(maint.deletedAt) : null },
   634→          create: {
   635→            id: maint.id, companyId, propertyId: maint.propertyId,
   636→            title: maint.title, description: maint.description,
   637→            status: maint.status || 'completed',
   638→            deletedAt: maint.deletedAt ? new Date(maint.deletedAt) : null,
   639→          },
   640→        })
   641→        summary.deletedMaintenance++
   642→      }
   643→    }
   644→
   645→    // ─── v2.0: Restore soft-deleted Recurring Bills ───
   646→    if (body.deleted?.recurringBills) {
   647→      for (const bill of body.deleted.recurringBills) {
   648→        await prisma.recurringBill.upsert({
   649→          where: { id: bill.id },
   650→          update: { providerName: bill.providerName, deletedAt: bill.deletedAt ? new Date(bill.deletedAt) : null },
   651→          create: {
   652→            id: bill.id, companyId, propertyId: bill.propertyId,
   653→            providerName: bill.providerName, serviceType: bill.serviceType,
   654→            currentOutstanding: bill.currentOutstanding,
   655→            totalAmountDue: bill.totalAmountDue,
   656→            nextDueDate: new Date(bill.nextDueDate),
   657→            status: bill.status || 'cancelled',
   658→            deletedAt: bill.deletedAt ? new Date(bill.deletedAt) : null,
   659→          },
   660→        })
   661→        summary.deletedRecurringBills++
   662→      }
   663→    }
   664→
   665→    // ─── v2.0: Restore soft-deleted Reservations ───
   666→    if (body.deleted?.reservations) {
   667→      for (const res of body.deleted.reservations) {
   668→        await prisma.reservation.upsert({
   669→          where: { id: res.id },
   670→          update: { status: res.status, deletedAt: res.deletedAt ? new Date(res.deletedAt) : null },
   671→          create: {
   672→            id: res.id, companyId, propertyId: res.propertyId,
   673→            prospectName: res.prospectName, prospectPhone: res.prospectPhone,
   674→            depositAmount: res.depositAmount, status: res.status || 'cancelled',
   675→            reservationDate: new Date(res.reservationDate),
   676→            deletedAt: res.deletedAt ? new Date(res.deletedAt) : null,
   677→          },
   678→        })
   679→        summary.deletedReservations++
   680→      }
   681→    }
   682→
   683→    // Log the restore completion
   684→    await createAuditLog({
   685→      action: 'RESTORE_COMPLETE',
   686→      entity: 'Company',
   687→      entityId: companyId,
   688→      userId: user.id,
   689→      companyId,
   690→      details: summary,
   691→    })
   692→
   693→    return successResponse({
   694→      message: 'Backup restored successfully',
   695→      summary,
   696→    })
   697→  } catch (error) {
   698→    console.error('Restore error:', error)
   699→    return errorResponse('Failed to restore backup', 500)
   700→  }
   701→}
