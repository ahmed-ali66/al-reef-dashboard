   142→   142→// ─── Expiry Warning ────────────────────────────────────────────────────────────
   143→   143→
   144→   144→function getDaysUntilExpiry(expiryDate: string | null): number | null {
   145→   145→  if (!expiryDate) return null
   146→   146→  const now = new Date()
   147→   147→  now.setHours(0, 0, 0, 0)
   148→   148→  const expiry = new Date(expiryDate)
   149→   149→  expiry.setHours(0, 0, 0, 0)
   150→   150→  return Math.ceil((expiry.getTime() - now.getTime()) / (1000 * 60 * 60 * 24))
   151→   151→}
