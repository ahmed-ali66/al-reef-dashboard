   152→  const getTenantPaymentStatus = (tenant: TenantData): 'paid' | 'partial' | 'overdue' | 'unpaid' | 'due-soon' => {
   153→    const payments = (tenant.payments || []).filter(p => p.month === selectedMonth && p.year === selectedYear)
   154→    const totalPaid = payments.reduce((sum, p) => sum + p.amount, 0)
   155→    const tenantAdjustments = getTenantAdjustments(tenant)

   157→    const creditApplied = Math.min(tenant.creditBalance || 0, Math.max(0, tenant.rentAmount - totalPaid - totalAdjustments))

   159→    if (totalCredits >= tenant.rentAmount) return 'paid'

   169→      if (dayOfMonth <= 4) return 'unpaid'
   170→      return 'overdue'

   174→    if (selectedYear < now.getFullYear() || (selectedYear === now.getFullYear() && selectedMonth < now.getMonth() + 1)) {
   175→      return 'overdue'
   176→    }
