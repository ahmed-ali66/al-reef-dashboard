   144→    if (a.status !== 'approved') return false
   145→    const startDate = new Date(a.effectiveYear, a.effectiveMonth - 1, 1)
   146→    const checkDate = new Date(year, month - 1, 1)

   149→      ((a.effectiveMonth - 1 + a.durationMonths) % 12),

   159→  const getTenantPaymentStatus = (tenant: TenantData): 'paid' | 'partial' | 'overdue' | 'unpaid' | 'due-soon' => {
   160→    // CRITICAL: Exclude HISTORICAL_DEBT payments from totalPaid because those payments
   161→    // are already reflected in the reduced tenant.openingBalance. Including them here
   162→    // would double-count the payment (once via reduced openingBalance, once via paymentsReceived).
   163→    const payments = (tenant.payments || []).filter(p => p.month === selectedMonth && p.year === selectedYear && p.allocationType !== 'HISTORICAL_DEBT')
   164→    const totalPaid = payments.reduce((sum, p) => sum + p.amount, 0)
   165→    const tenantAdjustments = getTenantAdjustments(tenant)
   166→    const totalAdjustments = tenantAdjustments.reduce((sum, a) => sum + a.amount, 0)
   167→    const creditApplied = Math.min(tenant.creditBalance || 0, Math.max(0, tenant.rentAmount - totalPaid - totalAdjustments))
   168→    const totalCredits = totalPaid + totalAdjustments + creditApplied
   169→    if (totalCredits >= tenant.rentAmount) return 'paid'
   170→    if (totalCredits > 0) return 'partial'

   173→    const now = new Date()

   176→    if (isSelectedCurrentMonth) {
   177→      const dayOfMonth = now.getDate()
   178→      if (dayOfMonth <= 2) return 'due-soon'
   179→      if (dayOfMonth <= 4) return 'unpaid'
   180→      return 'overdue'
   181→    }

   184→    if (selectedYear < now.getFullYear() || (selectedYear === now.getFullYear() && selectedMonth < now.getMonth() + 1)) {
   185→      return 'overdue'
   186→    }

   189→    return 'due-soon'
