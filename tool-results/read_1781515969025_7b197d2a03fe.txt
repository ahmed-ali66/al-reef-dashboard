   720→function cleanPhoneNumber(phone: string): string {
   721→  let cleanPhone = phone.replace(/[^0-9]/g, '')
   722→  if (cleanPhone.startsWith('0')) {
   723→    cleanPhone = '971' + cleanPhone.substring(1)
   724→  }
   725→  if (cleanPhone.startsWith('00')) {
   726→    cleanPhone = cleanPhone.substring(2)
   727→  }
   728→  if (!cleanPhone.startsWith('971') && !cleanPhone.startsWith('1') && !cleanPhone.startsWith('44') && !cleanPhone.startsWith('91') && !cleanPhone.startsWith('92') && !cleanPhone.startsWith('880')) {
   729→    cleanPhone = '971' + cleanPhone
   730→  }
   731→  return cleanPhone
   732→}
