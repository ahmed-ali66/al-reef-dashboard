     1→'use client'
     2→
     3→import { useState } from 'react'
     4→import { useAppStore } from '@/lib/store'
     5→import { useDataStore } from '@/lib/data-store'
     6→import type { LocalUser, ResetRequest } from '@/lib/data-store'
     7→import { t, getNameByLang } from '@/lib/i18n'
     8→import type { Language } from '@/lib/i18n'
     9→import { Button } from '@/components/ui/button'
    10→import { Input } from '@/components/ui/input'
    11→import { Label } from '@/components/ui/label'
    12→import {
    13→  Dialog,
    14→  DialogContent,
    15→  DialogHeader,
    16→  DialogTitle,
    17→  DialogFooter,
    18→  DialogDescription,
    19→} from '@/components/ui/dialog'
    20→import {
    21→  Select,
    22→  SelectContent,
    23→  SelectItem,
    24→  SelectTrigger,
    25→  SelectValue,
    26→} from '@/components/ui/select'
    27→import {
    28→  Users,
    29→  Plus,
    30→  Pencil,
    31→  Trash2,
    32→  KeyRound,
    33→  Copy,
    34→  Banknote,
    35→  Eye,
    36→  EyeOff,
    37→  ShieldCheck,
    38→  Shield,
    39→  User,
    40→  RefreshCw,
    41→  AlertTriangle,
    42→  CheckCircle2,
    43→  Bell,
    44→  BellRing,
    45→  Clock,
    46→  Check,
    47→  X,
    48→  MessageSquare,
    49→} from 'lucide-react'
    50→
    51→export default function UserManagement() {
    52→  const { language, authUser } = useAppStore()
    53→  const { users, resetRequests, addUser, updateUser, deleteUser, resetUserPassword, generateRandomPassword, resolveResetRequest, dismissResetRequest } = useDataStore()
    54→  const lang = language as Language
    55→
    56→  const [showAddDialog, setShowAddDialog] = useState(false)
    57→  const [showEditDialog, setShowEditDialog] = useState(false)
    58→  const [showDeleteDialog, setShowDeleteDialog] = useState(false)
    59→  const [showResetDialog, setShowResetDialog] = useState(false)
    60→  const [showCredentialsDialog, setShowCredentialsDialog] = useState(false)
    61→  const [selectedUser, setSelectedUser] = useState<LocalUser | null>(null)
    62→  const [visiblePasswords, setVisiblePasswords] = useState<Record<string, boolean>>({})
    63→
    64→  // Active tab: 'users' or 'requests'
    65→  const [activeTab, setActiveTab] = useState<'users' | 'requests'>('users')
    66→
    67→  // Add user form
    68→  const [newName, setNewName] = useState('')
    69→  const [newNameAr, setNewNameAr] = useState('')
    70→  const [newNameBn, setNewNameBn] = useState('')
    71→  const [newNameUr, setNewNameUr] = useState('')
    72→  const [newEmail, setNewEmail] = useState('')
    73→  const [newRole, setNewRole] = useState<'owner' | 'admin' | 'staff' | 'accountant'>('staff')
    74→  const [newPassword, setNewPassword] = useState('')
    75→  const [createdCredentials, setCreatedCredentials] = useState<{ email: string; password: string; name: string } | null>(null)
    76→
    77→  // Edit user form
    78→  const [editName, setEditName] = useState('')
    79→  const [editNameAr, setEditNameAr] = useState('')
    80→  const [editNameBn, setEditNameBn] = useState('')
    81→  const [editNameUr, setEditNameUr] = useState('')
    82→  const [editEmail, setEditEmail] = useState('')
    83→  const [editRole, setEditRole] = useState<'owner' | 'admin' | 'staff' | 'accountant'>('staff')
    84→
    85→  // Reset password
    86→  const [resetNewPassword, setResetNewPassword] = useState('')
    87→  const [resetCredentials, setResetCredentials] = useState<{ email: string; password: string } | null>(null)
    88→
    89→  const pendingRequests = resetRequests.filter(r => r.status === 'pending')
    90→  const resolvedRequests = resetRequests.filter(r => r.status !== 'pending')
    91→
    92→  const getRoleIcon = (role: string) => {
    93→    if (role === 'admin') return <Shield className="w-4 h-4 text-purple-600" />
    94→    if (role === 'owner') return <ShieldCheck className="w-4 h-4 text-emerald-600" />
    95→    if (role === 'accountant') return <Banknote className="w-4 h-4 text-blue-600" />
    96→    return <User className="w-4 h-4 text-muted-foreground" />
    97→  }
    98→
    99→  const getRoleLabel = (role: string) => {
   100→    if (role === 'owner') return t('ownerRole', lang)
   101→    if (role === 'admin') return t('adminRole', lang)
   102→    if (role === 'accountant') return t('accountantRole', lang)
   103→    return t('staffRole', lang)
   104→  }
   105→
   106→  const getRoleBadgeClass = (role: string) => {
   107→    if (role === 'admin') return 'bg-purple-100 text-purple-700 border-purple-200'
   108→    if (role === 'owner') return 'bg-emerald-100 text-emerald-700 border-emerald-200'
   109→    if (role === 'accountant') return 'bg-blue-100 text-blue-700 border-blue-200'
   110→    return 'bg-gray-100 text-gray-700 border-gray-200'
   111→  }
   112→
   113→  const [addError, setAddError] = useState('')
   114→  const [editError, setEditError] = useState('')
   115→
   116→  const handleAddUser = async () => {
   117→    if (users.some(u => u.email === newEmail)) {
   118→      alert(t('emailAlreadyExists', lang))
   119→      return
   120→    }
   121→
   122→    setAddError('')
   123→    const password = newPassword || generateRandomPassword()
   124→
   125→    try {
   126→      await addUser({
   127→        email: newEmail,
   128→        password,
   129→        name: newName,
   130→        nameAr: newNameAr,
   131→        nameBn: newNameBn,
   132→        nameUr: newNameUr,
   133→        role: newRole,
   134→        companyId: 'company-1',
   135→      })
   136→
   137→      setCreatedCredentials({ email: newEmail, password, name: newName })
   138→      setShowCredentialsDialog(true)
   139→      setShowAddDialog(false)
   140→      resetAddForm()
   141→    } catch (error: any) {
   142→      setAddError(error.message || (lang === 'en' ? 'Failed to create user' : 'فشل إنشاء المستخدم'))
   143→    }
   144→  }
   145→
   146→  const resetAddForm = () => {
   147→    setNewName('')
   148→    setNewNameAr('')
   149→    setNewNameBn('')
   150→    setNewNameUr('')
   151→    setNewEmail('')
   152→    setNewRole('staff')
   153→    setNewPassword('')
   154→    setAddError('')
   155→  }
   156→
   157→  const handleEditUser = async () => {
   158→    if (!selectedUser) return
   159→    if (editEmail !== selectedUser.email && users.some(u => u.email === editEmail)) {
   160→      alert(t('emailAlreadyExists', lang))
   161→      return
   162→    }
   163→    setEditError('')
   164→    try {
   165→      await updateUser(selectedUser.id, {
   166→        name: editName,
   167→        nameAr: editNameAr,
   168→        nameBn: editNameBn,
   169→        nameUr: editNameUr,
   170→        email: editEmail,
   171→        role: editRole,
   172→      })
   173→      setShowEditDialog(false)
   174→      setSelectedUser(null)
   175→    } catch (error: any) {
   176→      setEditError(error.message || (lang === 'en' ? 'Failed to update user' : 'فشل تحديث المستخدم'))
   177→    }
   178→  }
   179→
   180→  const handleDeleteUser = () => {
   181→    if (!selectedUser) return
   182→    // Admin can delete anyone except themselves
   183→    if (selectedUser.id === authUser?.id) {
   184→      alert(lang === 'en' ? 'You cannot delete your own account' : lang === 'ar' ? 'لا يمكنك حذف حسابك الخاص' : lang === 'bn' ? 'আপনি নিজের অ্যাকাউন্ট মুছতে পারবেন না' : 'آپ اپنا اکاؤنٹ حذف نہیں کر سکتے')
   185→      return
   186→    }
   187→    deleteUser(selectedUser.id)
   188→    setShowDeleteDialog(false)
   189→    setSelectedUser(null)
   190→  }
   191→
   192→  const handleResetPassword = () => {
   193→    if (!selectedUser) return
   194→    const password = resetNewPassword || generateRandomPassword()
   195→    resetUserPassword(selectedUser.id, password)
   196→    setResetCredentials({ email: selectedUser.email, password })
   197→    setShowResetDialog(false)
   198→    setShowCredentialsDialog(true)
   199→    setResetNewPassword('')
   200→    setSelectedUser(null)
   201→  }
   202→
   203→  const handleResolveRequest = (request: ResetRequest) => {
   204→    // Find if user exists with this email
   205→    const existingUser = users.find(u => u.email === request.email)
   206→    if (existingUser) {
   207→      // Open reset password dialog for this user
   208→      setSelectedUser(existingUser)
   209→      setResetNewPassword('')
   210→      setShowResetDialog(true)
   211→      // Mark request as resolved
   212→      resolveResetRequest(request.id, authUser?.name || 'admin')
   213→    } else {
   214→      // User not found - open add user dialog with email pre-filled
   215→      setNewName(request.name)
   216→      setNewEmail(request.email)
   217→      setNewNameAr('')
   218→      setNewNameBn('')
   219→      setNewNameUr('')
   220→      setNewRole('staff')
   221→      setNewPassword('')
   222→      setShowAddDialog(true)
   223→      // Mark request as resolved
   224→      resolveResetRequest(request.id, authUser?.name || 'admin')
   225→    }
   226→  }
   227→
   228→  const handleDismissRequest = (request: ResetRequest) => {
   229→    dismissResetRequest(request.id, authUser?.name || 'admin')
   230→  }
   231→
   232→  const copyToClipboard = (text: string) => {
   233→    navigator.clipboard.writeText(text)
   234→  }
   235→
   236→  const togglePasswordVisibility = (userId: string) => {
   237→    setVisiblePasswords(prev => ({ ...prev, [userId]: !prev[userId] }))
   238→  }
   239→
   240→  const formatDate = (dateStr: string) => {
   241→    const d = new Date(dateStr)
   242→    return d.toLocaleDateString(lang === 'ar' ? 'ar-AE' : lang === 'bn' ? 'bn-BD' : lang === 'ur' ? 'ur-PK' : 'en-AE', {
   243→      year: 'numeric',
   244→      month: 'short',
   245→      day: 'numeric',
   246→      hour: '2-digit',
   247→      minute: '2-digit',
   248→    })
   249→  }
   250→
   251→  return (
   252→    <div className="space-y-6">
   253→      {/* Header */}
   254→      <div className="flex items-center justify-between">
   255→        <div>
   256→          <h1 className="text-2xl font-bold flex items-center gap-2">
   257→            <Users className="w-7 h-7 text-deep-teal" />
   258→            {t('userManagement', lang)}
   259→          </h1>
   260→          <p className="text-muted-foreground text-sm mt-1">
   261→            {users.length} {t('usersCount', lang)}
   262→          </p>
   263→        </div>
   264→        <Button
   265→          onClick={() => {
   266→            resetAddForm()
   267→            setShowAddDialog(true)
   268→          }}
   269→          className="bg-deep-teal hover:bg-deep-teal/90 text-white"
   270→        >
   271→          <Plus className="w-4 h-4 mr-2" />
   272→          {t('addNewUser', lang)}
   273→        </Button>
   274→      </div>
   275→
   276→      {/* Tabs: Users / Reset Requests */}
   277→      <div className="flex gap-1 bg-muted/50 p-1 rounded-lg w-fit">
   278→        <button
   279→          onClick={() => setActiveTab('users')}
   280→          className={`px-4 py-2 rounded-md text-sm font-medium transition-all flex items-center gap-2 ${
   281→            activeTab === 'users'
   282→              ? 'bg-white text-foreground shadow-sm'
   283→              : 'text-muted-foreground hover:text-foreground'
   284→          }`}
   285→        >
   286→          <Users className="w-4 h-4" />
   287→          {lang === 'en' ? 'Users' : lang === 'ar' ? 'المستخدمون' : lang === 'bn' ? 'ব্যবহারকারী' : 'صارفین'}
   288→        </button>
   289→        <button
   290→          onClick={() => setActiveTab('requests')}
   291→          className={`px-4 py-2 rounded-md text-sm font-medium transition-all flex items-center gap-2 ${
   292→            activeTab === 'requests'
   293→              ? 'bg-white text-foreground shadow-sm'
   294→              : 'text-muted-foreground hover:text-foreground'
   295→          }`}
   296→        >
   297→          {pendingRequests.length > 0 ? (
   298→            <BellRing className="w-4 h-4 text-amber-600" />
   299→          ) : (
   300→            <Bell className="w-4 h-4" />
   301→          )}
   302→          {lang === 'en' ? 'Reset Requests' : lang === 'ar' ? 'طلبات إعادة التعيين' : lang === 'bn' ? 'রিসেট অনুরোধ' : 'ری سیٹ کی درخواستیں'}
   303→          {pendingRequests.length > 0 && (
   304→            <span className="bg-amber-500 text-white text-xs font-bold px-1.5 py-0.5 rounded-full min-w-[20px] text-center">
   305→              {pendingRequests.length}
   306→            </span>
   307→          )}
   308→        </button>
   309→      </div>
   310→
   311→      {/* USERS TAB */}
   312→      {activeTab === 'users' && (
   313→        <>
   314→          {/* Info Banner */}
   315→          <div className="bg-amber-50 border border-amber-200 rounded-xl p-4">
   316→            <div className="flex items-start gap-3">
   317→              <AlertTriangle className="w-5 h-5 text-amber-600 mt-0.5 shrink-0" />
   318→              <p className="text-sm text-amber-800">
   319→                {t('shareCredentials', lang)}
   320→              </p>
   321→            </div>
   322→          </div>
   323→
   324→          {/* Users Table */}
   325→          <div className="bg-white rounded-xl border border-border shadow-sm overflow-hidden">
   326→            <div className="overflow-x-auto">
   327→              <table className="w-full">
   328→                <thead>
   329→                  <tr className="border-b border-border bg-muted/30">
   330→                    <th className="text-left px-4 py-3 text-sm font-semibold text-muted-foreground">{t('roleName', lang)}</th>
   331→                    <th className="text-left px-4 py-3 text-sm font-semibold text-muted-foreground">{t('username', lang)}</th>
   332→                    <th className="text-left px-4 py-3 text-sm font-semibold text-muted-foreground">{t('password', lang)}</th>
   333→                    <th className="text-left px-4 py-3 text-sm font-semibold text-muted-foreground">{t('role', lang)}</th>
   334→                    <th className="text-right px-4 py-3 text-sm font-semibold text-muted-foreground">{t('actions', lang)}</th>
   335→                  </tr>
   336→                </thead>
   337→                <tbody>
   338→                  {users.map((user, index) => (
   339→                    <tr
   340→                      key={user.id}
   341→                      className={`border-b border-border last:border-0 ${index % 2 === 0 ? 'bg-white' : 'bg-muted/20'} hover:bg-muted/30 transition-colors`}
   342→                    >
   343→                      <td className="px-4 py-3">
   344→                        <div className="flex items-center gap-2">
   345→                          {getRoleIcon(user.role)}
   346→                          <div>
   347→                            <p className="font-medium text-sm">{getNameByLang(user, lang)}</p>
   348→                            {user.nameAr && (
   349→                              <p className="text-xs text-muted-foreground" dir="rtl">{user.nameAr}</p>
   350→                            )}
   351→                          </div>
   352→                        </div>
   353→                      </td>
   354→                      <td className="px-4 py-3">
   355→                        <code className="text-sm bg-muted px-2 py-0.5 rounded font-mono">{user.email}</code>
   356→                      </td>
   357→                      <td className="px-4 py-3">
   358→                        <div className="flex items-center gap-2">
   359→                          <code className="text-sm bg-muted px-2 py-0.5 rounded font-mono">
   360→                            {visiblePasswords[user.id] ? user.password : '••••••••'}
   361→                          </code>
   362→                          <button
   363→                            onClick={() => togglePasswordVisibility(user.id)}
   364→                            className="p-1 hover:bg-muted rounded transition-colors"
   365→                            title={visiblePasswords[user.id] ? t('hidePassword', lang) : t('showPassword', lang)}
   366→                          >
   367→                            {visiblePasswords[user.id] ? <EyeOff className="w-3.5 h-3.5" /> : <Eye className="w-3.5 h-3.5" />}
   368→                          </button>
   369→                          <button
   370→                            onClick={() => copyToClipboard(user.password)}
   371→                            className="p-1 hover:bg-muted rounded transition-colors"
   372→                            title={t('copyPassword', lang)}
   373→                          >
   374→                            <Copy className="w-3.5 h-3.5" />
   375→                          </button>
   376→                        </div>
   377→                      </td>
   378→                      <td className="px-4 py-3">
   379→                        <span className={`inline-flex items-center gap-1 px-2 py-1 rounded-full text-xs font-medium border ${getRoleBadgeClass(user.role)}`}>
   380→                          {getRoleIcon(user.role)}
   381→                          {getRoleLabel(user.role)}
   382→                        </span>
   383→                      </td>
   384→                      <td className="px-4 py-3">
   385→                        <div className="flex items-center justify-end gap-1">
   386→                          <button
   387→                            onClick={() => {
   388→                              setSelectedUser(user)
   389→                              setEditName(user.name)
   390→                              setEditNameAr(user.nameAr)
   391→                              setEditNameBn(user.nameBn)
   392→                              setEditNameUr(user.nameUr)
   393→                              setEditEmail(user.email)
   394→                              setEditRole(user.role)
   395→                              setShowEditDialog(true)
   396→                            }}
   397→                            className="p-2 hover:bg-muted rounded-lg transition-colors"
   398→                            title={t('editUser', lang)}
   399→                          >
   400→                            <Pencil className="w-4 h-4 text-blue-600" />
   401→                          </button>
   402→                          <button
   403→                            onClick={() => {
   404→                              setSelectedUser(user)
   405→                              setResetNewPassword('')
   406→                              setShowResetDialog(true)
   407→                            }}
   408→                            className="p-2 hover:bg-muted rounded-lg transition-colors"
   409→                            title={t('resetPassword', lang)}
   410→                          >
   411→                            <KeyRound className="w-4 h-4 text-amber-600" />
   412→                          </button>
   413→                          {user.id !== authUser?.id && (
   414→                            <button
   415→                              onClick={() => {
   416→                                setSelectedUser(user)
   417→                                setShowDeleteDialog(true)
   418→                              }}
   419→                              className="p-2 hover:bg-muted rounded-lg transition-colors"
   420→                              title={t('deleteUser', lang)}
   421→                            >
   422→                              <Trash2 className="w-4 h-4 text-red-600" />
   423→                            </button>
   424→                          )}
   425→                        </div>
   426→                      </td>
   427→                    </tr>
   428→                  ))}
   429→                </tbody>
   430→              </table>
   431→            </div>
   432→          </div>
   433→        </>
   434→      )}
   435→
   436→      {/* RESET REQUESTS TAB */}
   437→      {activeTab === 'requests' && (
   438→        <div className="space-y-4">
   439→          {/* Pending Requests */}
   440→          {pendingRequests.length > 0 && (
   441→            <div className="space-y-3">
   442→              <h3 className="text-lg font-semibold flex items-center gap-2">
   443→                <BellRing className="w-5 h-5 text-amber-600" />
   444→                {lang === 'en' ? 'Pending Requests' : lang === 'ar' ? 'الطلبات المعلقة' : lang === 'bn' ? 'অপেক্ষমাণ অনুরোধ' : 'زیر التواء درخواستیں'}
   445→                <span className="bg-amber-100 text-amber-700 text-xs font-bold px-2 py-0.5 rounded-full">
   446→                  {pendingRequests.length}
   447→                </span>
   448→              </h3>
   449→              {pendingRequests.map((request) => (
   450→                <div key={request.id} className="bg-white rounded-xl border-2 border-amber-200 shadow-sm p-4">
   451→                  <div className="flex items-start justify-between gap-4">
   452→                    <div className="flex-1 min-w-0">
   453→                      <div className="flex items-center gap-2 mb-2">
   454→                        <div className="w-8 h-8 rounded-full bg-amber-100 flex items-center justify-center shrink-0">
   455→                          <User className="w-4 h-4 text-amber-600" />
   456→                        </div>
   457→                        <div>
   458→                          <p className="font-medium text-sm">{request.name || lang === 'en' ? 'Unknown User' : lang === 'ar' ? 'مستخدم غير معروف' : lang === 'bn' ? 'অজানা ব্যবহারকারী' : 'نامعلوم صارف'}</p>
   459→                          <p className="text-xs text-muted-foreground">{request.email}</p>
   460→                        </div>
   461→                      </div>
   462→                      {request.message && (
   463→                        <div className="flex items-start gap-2 mb-2 ml-10">
   464→                          <MessageSquare className="w-3.5 h-3.5 text-muted-foreground mt-0.5 shrink-0" />
   465→                          <p className="text-sm text-muted-foreground">{request.message}</p>
   466→                        </div>
   467→                      )}
   468→                      <div className="flex items-center gap-2 ml-10 text-xs text-muted-foreground">
   469→                        <Clock className="w-3 h-3" />
   470→                        {formatDate(request.createdAt)}
   471→                      </div>
   472→                    </div>
   473→                    <div className="flex items-center gap-2 shrink-0">
   474→                      <Button
   475→                        size="sm"
   476→                        onClick={() => handleResolveRequest(request)}
   477→                        className="bg-emerald-600 hover:bg-emerald-700 text-white"
   478→                      >
   479→                        <KeyRound className="w-3.5 h-3.5 mr-1.5" />
   480→                        {lang === 'en' ? 'Reset & Resolve' : lang === 'ar' ? 'إعادة تعيين وحل' : lang === 'bn' ? 'রিসেট ও সমাধান' : 'ری سیٹ اور حل'}
   481→                      </Button>
   482→                      <Button
   483→                        size="sm"
   484→                        variant="outline"
   485→                        onClick={() => handleDismissRequest(request)}
   486→                        className="text-muted-foreground"
   487→                      >
   488→                        <X className="w-3.5 h-3.5 mr-1.5" />
   489→                        {lang === 'en' ? 'Dismiss' : lang === 'ar' ? 'رفض' : lang === 'bn' ? 'বাতিল' : 'مسترد'}
   490→                      </Button>
   491→                    </div>
   492→                  </div>
   493→                </div>
   494→              ))}
   495→            </div>
   496→          )}
   497→
   498→          {/* Resolved/Dismissed Requests */}
   499→          {resolvedRequests.length > 0 && (
   500→            <div className="space-y-3">
