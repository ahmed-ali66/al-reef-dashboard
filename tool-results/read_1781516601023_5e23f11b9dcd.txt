   146→   146→// Helper: check if an adjustment applies to a given month/year
   147→   147→// For multi-month adjustments (durationMonths > 1), the adjustment applies to every month
   148→   148→// from effectiveMonth/effectiveYear for durationMonths months
   149→   149→function isAdjustmentActiveInMonth(adjustment: RentAdjustmentData, month: number, year: number): boolean {
   150→   150→  if (adjustment.status !== 'approved') return false
   151→   151→
   152→   152→  const startDate = new Date(adjustment.effectiveYear, adjustment.effectiveMonth - 1, 1)
   153→   153→  const checkDate = new Date(year, month - 1, 1)
   154→   154→  const endDate = new Date(
   155→   155→    adjustment.effectiveYear + Math.floor((adjustment.effectiveMonth - 1 + adjustment.durationMonths) / 12),
   156→   156→    ((adjustment.effectiveMonth - 1 + adjustment.durationMonths) % 12),
   157→   157→    0 // last day of the end month
   158→   158→  )
   159→   159→
   160→   160→  return checkDate >= startDate && checkDate <= endDate
   161→   161→}

   163→   163→// PHASE 2: Dedup guard for 401 handling — prevent multiple simultaneous signOut calls
   164→   164→let isHandling401 = false
   165→   165→
   166→   166→// API helper — handles 401 by redirecting to login (with dedup guard)
   167→   167→async function apiCall(url: string, options?: RequestInit) {
   168→   168→  const res = await fetch(url, {
   169→   169→    ...options,
   170→   170→    headers: {
   171→   171→      'Content-Type': 'application/json',
   172→   172→      ...options?.headers,
   173→   173→    },
   174→   174→  })
   175→   175→
   176→   176→  // PHASE 2 FIX: On 401 Unauthorized, redirect to login page with dedup guard
   177→   177→  if (res.status === 401) {
   178→   178→    if (!isHandling401 && typeof window !== 'undefined') {
   179→   179→      isHandling401 = true
   180→   180→      try {
   181→   181→        // Use NextAuth signOut to properly clear session
   182→   182→        const { signOut } = await import('next-auth/react')
   183→   183→        await signOut({ callbackUrl: '/' })
   184→   184→      } finally {
   185→   185→        // Reset after a delay to allow future 401s if session expires again
   186→   186→        setTimeout(() => { isHandling401 = false }, 5000)
   187→   187→      }
   188→   188→    }
   189→   189→    throw new Error('Session expired. Please log in again.')
   190→   190→  }
   191→   191→
   192→   192→  // PHASE 2: Handle 409 Conflict (optimistic concurrency failure)
   193→   193→  if (res.status === 409) {
   194→   194→    const data = await res.json().catch(() => ({ error: 'Record was modified by another user' }))
   195→   195→    const error = new Error(data.error || 'Record was modified by another user. Please refresh and try again.')
   196→   196→    ;(error as any).isConflict = true
   197→   197→    ;(error as any).statusCode = 409
   198→   198→    throw error
   199→   199→  }
   200→   200→
   201→   201→  if (!res.ok) {
   202→   202→    const data = await res.json().catch(() => ({ error: 'Request failed' }))
   203→   203→    throw new Error(data.error || `Request failed with status ${res.status}`)
   204→   204→  }
   205→   205→  return res.json()
   206→   206→}
