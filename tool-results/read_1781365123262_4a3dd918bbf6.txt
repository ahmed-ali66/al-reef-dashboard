   129→  // Helper: check if adjustment is active in a given month
   130→  const isAdjustmentActiveInMonth = (a: RentAdjustmentData, month: number, year: number): boolean => {
   131→    if (a.status !== 'approved') return false
   132→    const startDate = new Date(a.effectiveYear, a.effectiveMonth - 1, 1)
   133→    const checkDate = new Date(year, month - 1, 1)
   134→    const endDate = new Date(
   135→      a.effectiveYear + Math.floor((a.effectiveMonth - 1 + a.durationMonths) / 12),
   136→      ((a.effectiveMonth - 1 + a.durationMonths) % 12),
   137→      0
   138→    )
   139→    return checkDate >= startDate && checkDate <= endDate
   140→  }

   143→    return (tenant.adjustments || []).filter(a => isAdjustmentActiveInMonth(a, selectedMonth, selectedYear))

   146→  const getTenantPaymentStatus = (tenant: TenantData): 'paid' | 'partial' | 'overdue' | 'unpaid' | 'due-soon' => {
   147→    const payments = (tenant.payments || []).filter(p => p.month === selectedMonth && p.year === selectedYear)
   148→    const totalPaid = payments.reduce((sum, p) => sum + p.amount, 0)
   149→    const tenantAdjustments = getTenantAdjustments(tenant)
   150→    const totalAdjustments = tenantAdjustments.reduce((sum, a) => sum + a.amount, 0)
   151→    const creditApplied = Math.min(tenant.creditBalance || 0, Math.max(0, tenant.rentAmount - totalPaid - totalAdjustments))
   152→    const totalCredits = totalPaid + totalAdjustments + creditApplied
   153→    if (totalCredits >= tenant.rentAmount) return 'paid'
   154→    if (totalCredits > 0) return 'partial'
   155→
   156→    // Calendar-based status for unpaid tenants
   157→    const now = new Date()
   158→    const isSelectedCurrentMonth = selectedMonth === now.getMonth() + 1 && selectedYear === now.getFullYear()
   159→
   160→    if (isSelectedCurrentMonth) {
   161→      const dayOfMonth = now.getDate()
   162→      if (dayOfMonth <= 2) return 'due-soon'
   163→      if (dayOfMonth <= 4) return 'unpaid'
   164→      return 'overdue'
   165→    }
   166→
   167→    // Past months - all unpaid are overdue
   168→    if (selectedYear < now.getFullYear() || (selectedYear === now.getFullYear() && selectedMonth < now.getMonth() + 1)) {
   169→      return 'overdue'
   170→    }
   171→
   172→    // Future months - due soon
   173→    return 'due-soon'
   174→  }
