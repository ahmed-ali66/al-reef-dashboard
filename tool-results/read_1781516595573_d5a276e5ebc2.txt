   146→// Helper: check if an adjustment applies to a given month/year
   147→// For multi-month adjustments (durationMonths > 1), the adjustment applies to every month
   148→// from effectiveMonth/effectiveYear for durationMonths months
   149→function isAdjustmentActiveInMonth(adjustment: RentAdjustmentData, month: number, year: number): boolean {
   150→  if (adjustment.status !== 'approved') return false
   151→
   152→  const startDate = new Date(adjustment.effectiveYear, adjustment.effectiveMonth - 1, 1)
   153→  const checkDate = new Date(year, month - 1, 1)
   154→  const endDate = new Date(
   155→    adjustment.effectiveYear + Math.floor((adjustment.effectiveMonth - 1 + adjustment.durationMonths) / 12),
   156→    ((adjustment.effectiveMonth - 1 + adjustment.durationMonths) % 12),
   157→    0 // last day of the end month
   158→  )
   159→
   160→  return checkDate >= startDate && checkDate <= endDate
   161→}

   163→// PHASE 2: Dedup guard for 401 handling — prevent multiple simultaneous signOut calls
   164→let isHandling401 = false
   165→
   166→// API helper — handles 401 by redirecting to login (with dedup guard)
   167→async function apiCall(url: string, options?: RequestInit) {
   168→  const res = await fetch(url, {
   169→    ...options,
   170→    headers: {
   171→      'Content-Type': 'application/json',
   172→      ...options?.headers,
   173→    },
   174→  })
   175→
   176→  // PHASE 2 FIX: On 401 Unauthorized, redirect to login page with dedup guard
   177→  if (res.status === 401) {
   178→    if (!isHandling401 && typeof window !== 'undefined') {
   179→      isHandling401 = true
   180→      try {
   181→        // Use NextAuth signOut to properly clear session
   182→        const { signOut } = await import('next-auth/react')
   183→        await signOut({ callbackUrl: '/' })
   184→      } finally {
   185→        // Reset after a delay to allow future 401s if session expires again
   186→        setTimeout(() => { isHandling401 = false }, 5000)
   187→      }
   188→    }
   189→    throw new Error('Session expired. Please log in again.')
   190→  }
   191→
   192→  // PHASE 2: Handle 409 Conflict (optimistic concurrency failure)
   193→  if (res.status === 409) {
   194→    const data = await res.json().catch(() => ({ error: 'Record was modified by another user' }))
   195→    const error = new Error(data.error || 'Record was modified by another user. Please refresh and try again.')
   196→    ;(error as any).isConflict = true
   197→    ;(error as any).statusCode = 409
   198→    throw error
   199→  }
   200→
   201→  if (!res.ok) {
   202→    const data = await res.json().catch(() => ({ error: 'Request failed' }))
   203→    throw new Error(data.error || `Request failed with status ${res.status}`)
   204→  }
   205→  return res.json()
   206→}

   731→    getDashboardData: () => {
   732→      const { company, tenants, payments, properties, expenses, maintenanceItems, adjustments } = get()
   733→      const now = new Date()
   734→      const currentMonth = now.getMonth() + 1
   735→      const currentYear = now.getFullYear()
   736→
   737→      const activeTenants = tenants.filter(t => isFinanciallyActive(t.status))
   738→      const expectedRevenue = activeTenants.reduce((sum, t) => sum + t.rentAmount, 0)
   739→      const currentMonthPayments = payments.filter(p => p.month === currentMonth && p.year === currentYear)
   740→      const collectedRevenue = currentMonthPayments.reduce((sum, p) => sum + p.amount, 0)
   741→
   742→      // Calculate adjustments for current month (including multi-month adjustments)
   743→      const currentMonthAdjustments = adjustments.filter(a => {
   744→        if (a.status !== 'approved') return false
   745→        return isAdjustmentActiveInMonth(a, currentMonth, currentYear)
   746→      })
   747→      const totalAdjustments = currentMonthAdjustments.reduce((sum, a) => sum + a.amount, 0)
   748→
   749→      const overdueTenants = activeTenants.filter(t => {
   750→        const hasPaid = payments.some(p => p.tenantId === t.id && p.month === currentMonth && p.year === currentYear)
   751→        return !hasPaid
   752→      })
   753→
   754→      const partialTenants = activeTenants.filter(t => {
   755→        const monthPayments = payments.filter(p => p.tenantId === t.id && p.month === currentMonth && p.year === currentYear)
   756→        const totalPaid = monthPayments.reduce((sum, p) => sum + p.amount, 0)
   757→        return totalPaid > 0 && totalPaid < t.rentAmount
   758→      })
   759→
   760→      const overdueAmount = overdueTenants.reduce((sum, t) => sum + t.rentAmount, 0)
   761→      const partialAmount = partialTenants.reduce((sum, t) => {

   763→        return sum + (t.rentAmount - paid)

   775→        const monthExpected = activeTenants.reduce((sum, t) => sum + t.rentAmount, 0)
   776→        const monthPayments = payments.filter(p => p.month === m && p.year === y)
   777→        const monthCollected = monthPayments.reduce((sum, p) => sum + p.amount, 0)
   778→        chartData.push({ month: monthNames[m - 1], expected: monthExpected, collected: monthCollected })
   779→      }
   780→
   781→      const recentPayments = payments
   782→        .sort((a, b) => new Date(b.date).getTime() - new Date(a.date).getTime())

   786→          tenant: tenants.find(t => t.id === p.tenantId) || undefined,

   789→      const totalExpenses = expenses.reduce((sum, e) => sum + e.amount, 0)

   795→          collectedRevenue,
   796→          overdueCount: overdueTenants.length,

   836→    getReportData: (month, year) => {
   837→      const { tenants, payments, properties, expenses, adjustments } = get()
   838→      const activeTenants = tenants.filter(t => isFinanciallyActive(t.status))
   839→      const expectedRevenue = activeTenants.reduce((sum, t) => sum + t.rentAmount, 0)
   840→      const monthPayments = payments.filter(p => p.month === month && p.year === year)
   841→      const totalRevenue = monthPayments.reduce((sum, p) => sum + p.amount, 0)
   842→
   843→      // Calculate adjustments for this month
   844→      const monthAdjustments = adjustments.filter(a => {
   845→        if (a.status !== 'approved') return false
   846→        return isAdjustmentActiveInMonth(a, month, year)
   847→      })
   848→      const totalAdjustments = monthAdjustments.reduce((sum, a) => sum + a.amount, 0)
   849→
   850→      const monthlyExpenses = expenses.filter(e => {
   851→        const d = new Date(e.date)
   852→        return d.getMonth() + 1 === month && d.getFullYear() === year
   853→      })
   854→      const totalExpenses = monthlyExpenses.reduce((sum, e) => sum + e.amount, 0)
   855→      const totalUnits = properties.reduce((sum, p) => sum + p.totalUnits, 0)
   856→      const occupiedUnits = activeTenants.length
   857→      const occupancyRate = totalUnits > 0 ? Math.round((occupiedUnits / totalUnits) * 100) : 0
   858→      const collectionRate = expectedRevenue > 0 ? Math.round((totalRevenue / expectedRevenue) * 100) : 0
   859→
   860→      const expenseBreakdown: Record<string, number> = {}
   861→      for (const e of monthlyExpenses) {
   862→        expenseBreakdown[e.category] = (expenseBreakdown[e.category] || 0) + e.amount
   863→      }
   864→
   865→      const vacantUnits = totalUnits - occupiedUnits
   866→      const avgRent = activeTenants.length > 0 ? expectedRevenue / activeTenants.length : 0
   867→      const vacancyLoss = vacantUnits * avgRent
   868→
   869→      const unpaidTenants = activeTenants.filter(t => !monthPayments.some(p => p.tenantId === t.id))
   870→      const partialTenants = activeTenants.filter(t => {
   871→        const paid = monthPayments.filter(p => p.tenantId === t.id).reduce((s, p) => s + p.amount, 0)
   872→        return paid > 0 && paid < t.rentAmount
   873→      })
   874→      const unpaidAmount = unpaidTenants.reduce((s, t) => s + t.rentAmount, 0) + partialTenants.reduce((s, t) => {
   875→        const paid = monthPayments.filter(p => p.tenantId === t.id).reduce((sum, p) => sum + p.amount, 0)
   876→        return s + (t.rentAmount - paid)
   877→      }, 0)
   878→
   879→      const trend: any[] = []
   880→      for (let i = 5; i >= 0; i--) {
   881→        let m = month - i, y = year
   882→        if (m <= 0) { m += 12; y -= 1 }
   883→        const tPayments = payments.filter(p => p.month === m && p.year === y)
   884→        const revenue = tPayments.reduce((s, p) => s + p.amount, 0)
   885→        const tExpenses = expenses.filter(e => {
   886→          const d = new Date(e.date)
   887→          return d.getMonth() + 1 === m && d.getFullYear() === y
   888→        }).reduce((s, e) => s + e.amount, 0)
   889→        trend.push({ month: m, year: y, revenue, expenses: tExpenses, profit: revenue - tExpenses })
   890→      }
   891→
   892→      const rentalIncome = totalRevenue
   893→      const otherIncome = 0
   894→      const grossRevenue = rentalIncome + otherIncome
   895→      const badDebt = unpaidAmount
   896→      const grossProfit = grossRevenue - vacancyLoss - badDebt
   897→      const costOfOperations = totalExpenses
   898→      const netIncome = grossProfit - costOfOperations
   899→
   900→      return {
   901→        month, year, totalRevenue, expectedRevenue, totalExpenses,
   902→        profitLoss: totalRevenue - totalExpenses,
   903→        occupancyRate, collectionRate, totalUnits, occupiedUnits,
   904→        expenseBreakdown, monthlyExpenses, trend,
   905→        rentalIncome, otherIncome, grossRevenue, vacancyLoss, badDebt,
   906→        grossProfit, costOfOperations, netIncome,
   907→        totalAdjustments,
   908→        cashCollected: totalRevenue,
   909→        adjustmentTotal: totalAdjustments,
   910→        netRevenue: totalRevenue + totalAdjustments,
   911→      }
   912→    },
